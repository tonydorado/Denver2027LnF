Google Sheets Backend Setup

Connect this front-end form directly to a private Google Sheet in 3 minutes.

  1. Create a fresh Google Sheet and name two tabs: Lost_Reports and Found_Inventory.
  2. Open Extensions > Apps Script in Google Sheets.
  3. Paste the Google Apps Script snippet below and click Deploy > New Deployment.
  4. Select type: Web app, set Who has access to "Anyone", and click Deploy.
  5. Copy the Web App URL and paste it below.
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  
  if (data.type === 'LOST') {
    var lostSheet = sheet.getSheetByName('Lost_Reports') || sheet.insertSheet('Lost_Reports');
    lostSheet.appendRow([
      new Date(), data.id, data.name, data.phone, data.email, 
      data.language, data.category, data.color, data.location, data.description, data.status
    ]);
  } else if (data.type === 'FOUND') {
    var foundSheet = sheet.getSheetByName('Found_Inventory') || sheet.insertSheet('Found_Inventory');
    foundSheet.appendRow([
      new Date(), data.tag, data.bin, data.category, data.color, 
      data.description, data.volunteer, data.location, data.status
    ]);
  }
  
  return ContentService.createTextOutput(JSON.stringify({result: 'success'}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet() {
  return ContentService.createTextOutput("Convention Lost & Found Backend Active");
}
